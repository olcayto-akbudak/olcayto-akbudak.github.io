SQL yöntemi

WITH v AS (SELECT sku, units*unit_cost AS stock_value FROM data), r AS (SELECT sku,stock_value,ROUND(100.0*SUM(stock_value) OVER (ORDER BY stock_value DESC)/SUM(stock_value) OVER (),1) AS cumulative_pct FROM v) SELECT CASE WHEN cumulative_pct<=80 THEN 'A' WHEN cumulative_pct<=95 THEN 'B' ELSE 'C' END AS abc, COUNT(*) AS urun, SUM(stock_value) AS stok_degeri FROM r GROUP BY abc ORDER BY abc;

Yorumlama sınırı

ABC sınıfları birikimli stok değerinin %80 ve %95 eşiklerine göre tanımlanır.

WITH stats AS (SELECT AVG(amount) AS mean, SQRT(AVG(amount*amount)-AVG(amount)*AVG(amount)) AS sigma FROM data) SELECT channel, COUNT(*) AS islem, SUM(amount>(SELECT mean+3*sigma FROM stats)) AS uc_sigma_ustu FROM data GROUP BY channel ORDER BY uc_sigma_ustu DESC;
