WITH r AS (SELECT payment_id,SUM(amount_minor) AS refund FROM refunds GROUP BY payment_id), s AS (SELECT payment_id,SUM(amount_minor) AS paid FROM settlements GROUP BY payment_id), d AS (SELECT p.*,gross_minor-fee_minor-COALESCE(refund,0) AS expected,COALESCE(paid,0) AS paid,CASE WHEN s.payment_id IS NULL THEN 'Bekliyor' WHEN ABS(COALESCE(paid,0)-(gross_minor-fee_minor-COALESCE(refund,0)))<=1 THEN 'Tam' WHEN paid<gross_minor-fee_minor-COALESCE(refund,0) THEN 'Eksik' ELSE 'Fazla' END AS classification FROM payments p LEFT JOIN r USING(payment_id) LEFT JOIN s USING(payment_id)) SELECT classification,COUNT(*) AS payments,ROUND(SUM((paid-expected)/100.0*try_rate),2) AS signed_difference_try,ROUND(SUM(ABS(paid-expected)/100.0*try_rate),2) AS absolute_difference_try FROM d JOIN fx USING(currency) GROUP BY classification;
